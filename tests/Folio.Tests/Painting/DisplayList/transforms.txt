=== a transform wraps the box, its descendants and its outline, mapped in canvas coordinates
<!DOCTYPE html><style>body { margin: 0 }</style><div style="margin-left: 10px; width: 20px; height: 10px; background: rgb(1, 0, 0); outline: 1px solid rgb(3, 0, 0); transform: translate(5px, 6px)"><div style="height: 5px; background: rgb(2, 0, 0)"></div></div>
---
transform 1,0,0,1,5,6
fill 10,0 20x10 rgb(1, 0, 0)
fill 10,0 20x5 rgb(2, 0, 0)
border 9,-1 22x12 1 solid rgb(3, 0, 0)
pop

=== rotation is about the centre of the border box by default
<!DOCTYPE html><style>body { margin: 0 }</style><div style="width: 20px; height: 10px; background: rgb(1, 0, 0); transform: rotate(90deg)"></div>
---
transform 0,1,-1,0,15,-5
fill 0,0 20x10 rgb(1, 0, 0)
pop

=== transform-origin is relative to the box wherever it is on the canvas
<!DOCTYPE html><style>body { margin: 0 }</style><div style="margin-left: 10px; width: 20px; height: 10px; background: rgb(1, 0, 0); transform-origin: 0 0; transform: scale(2)"></div>
---
transform 2,0,0,2,-10,0
fill 10,0 20x10 rgb(1, 0, 0)
pop

=== the individual properties combine with transform into one matrix
<!DOCTYPE html><style>body { margin: 0 }</style><div style="width: 20px; height: 10px; background: rgb(1, 0, 0); transform-origin: 0 0; translate: 10px; scale: 2; transform: translateY(1px)"></div>
---
transform 2,0,0,2,10,2
fill 0,0 20x10 rgb(1, 0, 0)
pop

=== nested transforms nest
<!DOCTYPE html><style>body { margin: 0 } div { width: 20px }</style><div style="height: 20px; background: rgb(1, 0, 0); transform: translate(5px)"><div style="height: 10px; background: rgb(2, 0, 0); transform: translate(0, 3px)"></div></div>
---
transform 1,0,0,1,5,0
fill 0,0 20x20 rgb(1, 0, 0)
transform 1,0,0,1,0,3
fill 0,0 20x10 rgb(2, 0, 0)
pop
pop

=== clips from outside stay outside the transform, the box's own overflow clip goes inside it
<!DOCTYPE html><style>body { margin: 0 }</style><div style="overflow: hidden; width: 50px; height: 50px"><div style="overflow: hidden; width: 20px; height: 20px; transform: translate(5px, 5px)"><div style="height: 40px; background: rgb(1, 0, 0)"></div></div></div>
---
clip 0,0 50x50
transform 1,0,0,1,5,5
clip 0,0 20x20
fill 0,0 20x40 rgb(1, 0, 0)
pop
pop
pop

=== opacity composites the transformed box
<!DOCTYPE html><style>body { margin: 0 }</style><div style="height: 10px; background: rgb(1, 0, 0); opacity: 0.5; transform: translate(1px)"></div>
---
transform 1,0,0,1,1,0
opacity 0.5

pop
pop

=== a transform that cannot be inverted hides the box and its content
<!DOCTYPE html><style>body { margin: 0 } div { height: 10px }</style><div style="background: rgb(1, 0, 0); transform: scale(1, 0)"><div style="position: relative; z-index: 1; background: rgb(2, 0, 0)"></div></div><div style="background: rgb(3, 0, 0)"></div>
---
fill 0,10 800x10 rgb(3, 0, 0)

=== non-atomic inline boxes are not transformed
<!DOCTYPE html><style>body { margin: 0 }</style><span style="transform: scale(2)">a</span>
---
glyphs 0,13 16px rgb(0, 0, 0)

=== inline blocks are
<!DOCTYPE html><style>body { margin: 0 }</style><span style="display: inline-block; width: 10px; height: 10px; background: rgb(1, 0, 0); transform: translate(2px)"></span>
---
transform 1,0,0,1,2,0
fill 0,3 10x10 rgb(1, 0, 0)
pop

=== a table is transformed once, by its wrapper
<!DOCTYPE html><style>body { margin: 0 } td { padding: 0; width: 10px; height: 10px }</style><table style="border-spacing: 0; background: rgb(1, 0, 0); outline: 1px solid rgb(2, 0, 0); transform: translate(2px)"><tr><td></td></tr></table>
---
transform 1,0,0,1,2,0
fill 0,0 10x10 rgb(1, 0, 0)
border -1,-1 12x12 1 solid rgb(2, 0, 0)
pop

=== shadows, text and its decorations paint inside the transform
<!DOCTYPE html><style>body { margin: 0 }</style><div style="width: 20px; height: 20px; box-shadow: 2px 2px rgb(1, 0, 0); text-decoration: underline; text-decoration-skip-ink: none; transform: translate(3px)">a</div>
---
transform 1,0,0,1,3,0
shadow 2,2 20x20 rgb(1, 0, 0) blur 0 outside 0,0 20x20
decoration 0,14.6 16x1 solid rgb(0, 0, 0)
glyphs 0,13 16px rgb(0, 0, 0)
pop

=== z-index does not apply to a transformed box that is not positioned
<!DOCTYPE html><style>body { margin: 0 } div { height: 10px }</style><div style="margin-bottom: -10px; background: rgb(1, 0, 0); z-index: -1; transform: translate(0)"></div><div style="background: rgb(2, 0, 0); z-index: 1; transform: translate(0)"></div><div style="position: relative; background: rgb(3, 0, 0)"></div>
---
transform 1,0,0,1,0,0
fill 0,0 800x10 rgb(1, 0, 0)
pop
transform 1,0,0,1,0,0
fill 0,0 800x10 rgb(2, 0, 0)
pop
fill 0,10 800x10 rgb(3, 0, 0)

=== fixed descendants paint inside the transform of the box they are positioned in
<!DOCTYPE html><style>body { margin: 0 }</style><div style="height: 10px; transform: translate(5px)"><div style="position: fixed; top: 0; left: 0; width: 5px; height: 5px; background: rgb(1, 0, 0)"></div></div>
---
transform 1,0,0,1,5,0
fill 0,0 5x5 rgb(1, 0, 0)
pop

=== a box under its parent's perspective is drawn through a projective transform
<!DOCTYPE html><style>body { margin: 0 } .p { perspective: 100px; perspective-origin: 0 0; width: 100px; height: 100px } .c { height: 50px; background: red; transform: rotateY(60deg); transform-origin: 0 0 }</style><div class="p"><div class="c"></div></div>
---
project 0.5,0,0.01,0,1,0,0,0,1
fill 0,0 100x50 rgb(255, 0, 0)
pop
