=== instances inherit from their use element, not from where the original is
<!DOCTYPE html><style>body { margin: 0 } svg { display: block }</style><svg width="200" height="100"><defs fill="green"><g id="icon"><path d="M0 0 H10 V10 Z" /><path d="M0 0 H1 V1 Z" fill="lime" /></g></defs><use href="#icon" fill="red" /><use xlink:href="#icon" fill="blue" x="20" /></svg>
---
transform 1,0,0,1,0,0
clip 0,0 200x100
fill path M0,0 L10,0 L10,10 Z rgb(255, 0, 0)
fill path M0,0 L1,0 L1,1 Z rgb(0, 255, 0)
transform 1,0,0,1,20,0
fill path M0,0 L10,0 L10,10 Z rgb(0, 0, 255)
fill path M0,0 L1,0 L1,1 Z rgb(0, 255, 0)
pop
pop
