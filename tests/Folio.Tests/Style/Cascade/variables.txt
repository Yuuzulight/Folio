=== var substitutes and custom properties inherit
<!DOCTYPE html><style>:root { --accent: #f00; --gap: 12px } p { color: var(--accent); margin-top: var(--gap) }</style><p id=a>x</p>
---
#a color: rgb(255, 0, 0)
#a margin-top: 12px

=== fallback when the property is missing
<!DOCTYPE html><style>p { color: var(--missing, blue); margin-top: var(--missing, var(--also-missing, 4px)) }</style><p id=a>x</p>
---
#a color: rgb(0, 0, 255)
#a margin-top: 4px

=== invalid at computed-value time acts as unset
<!DOCTYPE html><style>div { color: green; margin-top: 7px } p { color: red; color: var(--missing); margin-top: var(--missing) }</style><div><p id=a>x</p></div>
---
#a color: rgb(0, 128, 0)
#a margin-top: 0px

=== var inside calc
<!DOCTYPE html><style>p { --n: 3; --w: 10px; width: calc(var(--n) * var(--w)) }</style><p id=a>x</p>
---
#a width: 30px

=== var in a shorthand
<!DOCTYPE html><style>p { --m: 1px 2px; margin: var(--m) }</style><p id=a>x</p>
---
#a margin-top: 1px
#a margin-right: 2px
#a margin-bottom: 1px

=== substituted tokens do not merge
<!DOCTYPE html><style>p { --n: 10; width: var(--n)px }</style><p id=a>x</p>
---
#a width: auto

=== custom properties referring to each other
<!DOCTYPE html><style>:root { --base: 4px; --double: calc(var(--base) * 2) } p { margin-top: var(--double) }</style><p id=a>x</p>
---
#a margin-top: 8px

=== cycles make every property in them invalid
<!DOCTYPE html><style>p { --a: var(--b, 1px); --b: var(--a, 2px); margin-top: var(--a, 5px); margin-left: var(--b, 6px) }</style><p id=a>x</p>
---
#a margin-top: 5px
#a margin-left: 6px

=== a child overrides a custom property for its subtree
<!DOCTYPE html><style>:root { --c: red } .x { --c: blue } p { color: var(--c) }</style><div class=x><p id=a>x</p></div><p id=b>y</p>
---
#a color: rgb(0, 0, 255)
#b color: rgb(255, 0, 0)

=== registered properties that do not inherit
<!DOCTYPE html><style>@property --size { syntax: '<length>'; inherits: false; initial-value: 3px } div { --size: 9px } p { margin-top: var(--size) }</style><div><p id=a>x</p></div>
---
#a margin-top: 3px

=== var in the style attribute
<!DOCTYPE html><style>:root { --c: blue }</style><p id=a style="color: var(--c)">x</p>
---
#a color: rgb(0, 0, 255)

=== registered lengths compute to px on the element that declares them
<!DOCTYPE html><style>@property --len { syntax: '<length>'; inherits: true; initial-value: 0px } div { font-size: 10px; --len: 2em } p { font-size: 20px; margin-top: var(--len) }</style><div id=d><p id=a>x</p></div>
---
#d --len: 20px
#a --len: 20px
#a margin-top: 20px

=== unregistered lengths are substituted as written
<!DOCTYPE html><style>div { font-size: 10px; --len: 2em } p { font-size: 20px; margin-top: var(--len) }</style><div><p id=a>x</p></div>
---
#a margin-top: 40px

=== a registered property has its initial value where nothing sets it
<!DOCTYPE html><style>@property --gap { syntax: '<length>'; inherits: true; initial-value: 1in } p { margin-top: var(--gap) }</style><p id=a>x</p>
---
#a --gap: 96px
#a margin-top: 96px

=== a value that does not match the syntax is inherited or initial
<!DOCTYPE html><style>@property --in { syntax: '<length>'; inherits: true; initial-value: 1px } @property --own { syntax: '<color>'; inherits: false; initial-value: red } div { --in: 5px } p { --in: blue; --own: 10px }</style><div><p id=a>x</p></div>
---
#a --in: 5px
#a --own: rgb(255, 0, 0)

=== invalid registrations register nothing
<!DOCTYPE html><style>@property --a { syntax: '<length>'; inherits: true } @property --b { syntax: '<length>'; inherits: true; initial-value: 1em } @property --c { syntax: '<size>'; inherits: true; initial-value: 1px } @property --d { syntax: '<length>'; initial-value: 1px } p { --a: 2em; --b: 2em; --c: 2em; --d: 2em }</style><p id=a>x</p>
---
#a --a: 2em
#a --b: 2em
#a --c: 2em
#a --d: 2em

=== registered syntaxes: alternatives, lists, keywords and other types
<!DOCTYPE html><style>@property --x { syntax: '<length> | auto'; inherits: false; initial-value: auto } @property --list { syntax: '<number>+'; inherits: false; initial-value: 0 } @property --commas { syntax: '<color>#'; inherits: false; initial-value: red } @property --t { syntax: '<time>'; inherits: false; initial-value: 0s } @property --turn { syntax: '<angle>'; inherits: false; initial-value: 0deg } @property --any { syntax: '*'; inherits: false } p { font-size: 10px; --x: 3em; --list: 1 2.5 3; --commas: blue, rgb(0 128 0); --t: 250ms; --turn: 0.5turn; --any: foo( bar ) }</style><p id=a>x</p>
---
#a --x: 30px
#a --list: 1 2.5 3
#a --commas: rgb(0, 0, 255), rgb(0, 128, 0)
#a --t: 0.25s
#a --turn: 180deg
#a --any: foo( bar )
