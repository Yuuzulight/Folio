=== floats shrink to fit their content
<!DOCTYPE html><style>body { margin: 0 }</style><div style="float: left">abc de</div>
---
html 0,0 800x16
  body 0,0 800x0
    div 0,0 96x16
      line 0,0 96x16
        text "abc " 0,0 64x16
        text "de" 64,0 32x16

=== a float wider than its space shrinks to the space but not below min-content
%viewport 60 600
<!DOCTYPE html><style>body { margin: 0 }</style><div style="float: left">abc de</div>
---
html 0,0 60x32
  body 0,0 60x0
    div 0,0 60x32
      line 0,0 60x16
        text "abc " 0,0 64x16
      line 0,16 60x16
        text "de" 0,16 32x16

=== inline-blocks shrink to fit and sit on the baseline
<!DOCTYPE html><style>body, p { margin: 0 }</style><p>a<span style="display: inline-block">bc</span>d</p>
---
html 0,0 800x16
  body 0,0 800x16
    p 0,0 800x16
      line 0,0 800x16
        text "a" 0,0 16x16
        span 16,0 32x16
          line 16,0 32x16
            text "bc" 16,0 32x16
        text "d" 48,0 16x16

=== absolutely positioned boxes shrink to fit unless both insets are set
<!DOCTYPE html><style>body { margin: 0 }</style><div style="position: absolute; left: 10px">abc</div><div style="position: absolute; top: 20px; left: 10px; right: 10px">abc</div>
---
html 0,0 800x0
  body 0,0 800x0
div 10,0 48x16
  line 10,0 48x16
    text "abc" 10,0 48x16
div 10,20 780x16
  line 10,20 780x16
    text "abc" 10,20 48x16

=== a block's intrinsic width is its widest child's
<!DOCTYPE html><style>body { margin: 0 }</style><div style="float: left"><div style="padding: 0 5px">abc</div><div style="width: 20px; margin-left: 30px"></div></div>
---
html 0,0 800x16
  body 0,0 800x0
    div 0,0 58x16
      div 0,0 58x16
        line 5,0 48x16
          text "abc" 5,0 48x16
      div 30,16 20x0

=== nowrap keeps atomic inlines together, and they count together in min-content
<!DOCTYPE html><style>body, div { margin: 0 }</style><div style="width: 60px; white-space: nowrap"><span style="display: inline-block; width: 20px; height: 20px"></span><span style="display: inline-block; width: 20px; height: 20px"></span><span style="display: inline-block; width: 20px; height: 20px"></span><span style="display: inline-block; width: 20px; height: 20px"></span><span style="display: inline-block; width: 20px; height: 20px"></span></div><div style="float: left; width: min-content; white-space: nowrap"><span style="display: inline-block; width: 20px; height: 20px"></span><span style="display: inline-block; width: 20px; height: 20px"></span><span style="display: inline-block; width: 20px; height: 20px"></span></div>
---

  body 0,0 800x23
    div 0,0 60x23

        span 20,0 20x20
        span 40,0 20x20
        span 60,0 20x20
        span 80,0 20x20
    div 0,23 60x23
      line 0,23 60x23
        span 0,23 20x20
